\subsection{Szimulációs MATLAB szkript}
A PID szabályozó és a mechanikai szakasz zárt köri vizsgálatát az \textbf{1. forráskód} szerinti MATLAB szkript végzi.

\begin{matlabbox}
\begin{lstlisting}[style=matlabdark]
% Szabalyozastechnika - 2. szorgalmi feladat
% Hallgato: Minta Janos (NEPTUN)

clear; clc; close all;

(*@\matlabrule@*)%% === 1. RÉSZ: Változók és Rendszerelemek ===

% Valtozok definialasa
syms s t Y_sym

% Rendszerelemek
W_p = 1 / (1 + 2*s);
W_c = 4 + 5/s;
W_v = 3;

% Bemeneti jelek
R   = 3/s;

(*@\matlabrule@*)%% === 2. RÉSZ: Zárt Kör és Dinamikai Szimuláció ===
s_num = tf('s');
P = 1 / (1 + 2*s_num);
C = 4 + 5/s_num;

% Zart kor es vizsgalatok
L = C * P;              % Felnyitott kor
T = feedback(L, 3);     % Zart szabalyozasi kor
S = feedback(1, L);     % Erzekenysegi fuggveny

figure('Name', 'Zart kor lepesvalasza');
step(T, 10);
grid on;
title('BME MOGI - Szabalyozasi kor lepesvalasza');
xlabel('Ido [s]'); ylabel('Kimenet y(t)');
\end{lstlisting}
\end{matlabbox}
